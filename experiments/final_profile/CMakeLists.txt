cmake_minimum_required(VERSION 3.16)
project(vflhcds_final_profile LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(VF_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
file(GLOB_RECURSE pinned_sources CONFIGURE_DEPENDS "${VF_ROOT}/src/*.cpp" "${VF_ROOT}/include/*.hpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  instrument.py observer.hpp source_pins.json ${pinned_sources} "${VF_ROOT}/CMakeLists.txt")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/instrument.py"
  --root "${VF_ROOT}" --output "${CMAKE_CURRENT_BINARY_DIR}/instrumented"
  RESULT_VARIABLE instrument_status)
if(NOT instrument_status EQUAL 0)
  message(FATAL_ERROR "Cannot instrument pinned production sources")
endif()
set(BUILD_TESTING OFF CACHE BOOL "Tests below include matched production probes" FORCE)
add_subdirectory("${VF_ROOT}" production)
get_target_property(original_sources vflhcds_lib SOURCES)
set(diagnostic_sources)
foreach(source IN LISTS original_sources)
  if(EXISTS "${CMAKE_CURRENT_BINARY_DIR}/instrumented/${source}")
    list(APPEND diagnostic_sources "${CMAKE_CURRENT_BINARY_DIR}/instrumented/${source}")
  elseif(IS_ABSOLUTE "${source}")
    list(APPEND diagnostic_sources "${source}")
  else()
    list(APPEND diagnostic_sources "${VF_ROOT}/${source}")
  endif()
endforeach()
add_library(final_profile_lib STATIC ${diagnostic_sources})
target_include_directories(final_profile_lib PUBLIC "${VF_ROOT}/include" "${CMAKE_CURRENT_SOURCE_DIR}")
target_include_directories(final_profile_lib SYSTEM PUBLIC "${VFLHCDS_BOOST_ROOT}")
target_link_libraries(final_profile_lib PRIVATE vflhcds_build_options)
add_executable(vflhcds_final_profile "${CMAKE_CURRENT_BINARY_DIR}/instrumented/src/cli/main.cpp")
target_link_libraries(vflhcds_final_profile PRIVATE final_profile_lib vflhcds_build_options)
add_executable(final_profile_timer_test timer_test.cpp)
target_link_libraries(final_profile_timer_test PRIVATE final_profile_lib vflhcds_build_options)
add_executable(final_profile_boundary_test boundary_test.cpp)
target_link_libraries(final_profile_boundary_test PRIVATE final_profile_lib vflhcds_build_options)
foreach(variant control diagnostic)
  if(variant STREQUAL "control")
    set(library vflhcds_lib)
  else()
    set(library final_profile_lib)
  endif()
  foreach(test primitives solver_probe core)
    if(test STREQUAL "primitives")
      set(source tests/m2_primitives.cpp)
    elseif(test STREQUAL "solver_probe")
      set(source tests/m3_probe.cpp)
    else()
      set(source tests/m4_core.cpp)
    endif()
    add_executable(final_${variant}_${test} "${VF_ROOT}/${source}")
    target_link_libraries(final_${variant}_${test} PRIVATE ${library} vflhcds_build_options)
    target_compile_definitions(final_${variant}_${test} PRIVATE VFLHCDS_TEST_COUNT_MEMBERSHIP)
  endforeach()
endforeach()
enable_testing()
add_test(NAME T18_exclusive_cleanup COMMAND final_profile_timer_test)
add_test(NAME T18_T20_T27_artifact_contracts COMMAND "${Python3_EXECUTABLE}" -m unittest discover
  -s "${CMAKE_CURRENT_SOURCE_DIR}" -p "test_*.py" -v)
add_test(NAME T18_T20_native_boundaries COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_native_boundaries.py"
  "$<TARGET_FILE:final_profile_boundary_test>")
foreach(variant control diagnostic)
  add_test(NAME T20_${variant}_numeric COMMAND final_${variant}_primitives)
  add_test(NAME T20_${variant}_core COMMAND final_${variant}_core)
endforeach()
