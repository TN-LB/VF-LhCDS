cmake_minimum_required(VERSION 3.16)
project(vflhcds_fine_profile LANGUAGES CXX)
get_filename_component(VF_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  instrument.py "${VF_ROOT}/src/clique/materialized.cpp" "${VF_ROOT}/src/oracle/closure.cpp")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/instrument.py"
  --root "${VF_ROOT}" --output "${CMAKE_CURRENT_BINARY_DIR}/instrumented"
  RESULT_VARIABLE instrument_status)
if(NOT instrument_status EQUAL 0)
  message(FATAL_ERROR "Cannot instrument the pinned production sources")
endif()
# Reuse the production source list/options in a separate build directory only.
set(BUILD_TESTING OFF CACHE BOOL "Root tests are exercised separately" FORCE)
add_subdirectory("${VF_ROOT}" production EXCLUDE_FROM_ALL)
get_target_property(original_sources vflhcds_lib SOURCES)
set(profile_sources)
foreach(source IN LISTS original_sources)
  if(source STREQUAL "src/clique/materialized.cpp" OR source STREQUAL "src/oracle/closure.cpp")
    get_filename_component(base "${source}" NAME)
    list(APPEND profile_sources "${CMAKE_CURRENT_BINARY_DIR}/instrumented/${base}")
  elseif(IS_ABSOLUTE "${source}")
    list(APPEND profile_sources "${source}")
  else()
    list(APPEND profile_sources "${VF_ROOT}/${source}")
  endif()
endforeach()
set_property(TARGET vflhcds_lib PROPERTY SOURCES "${profile_sources}")
target_include_directories(vflhcds_lib PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_executable(vflhcds_fine_profile main.cpp)
target_link_libraries(vflhcds_fine_profile PRIVATE vflhcds::lib vflhcds_build_options)
add_executable(profile_timer_test timer_test.cpp)
target_link_libraries(profile_timer_test PRIVATE vflhcds_build_options)
enable_testing()
add_test(NAME exclusive_profile_timer COMMAND profile_timer_test)
